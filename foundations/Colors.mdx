  white: '#FFFFFF', black: '#000000', 'gray-50': '#F4F4F4', 'gray-100': '#E9E9E9', 'gray-600': '#666666',

  'red-dark': '#A2191F', red: '#DA1E28', 'red-tint': '#FFF1F1', green: '#84FF00', 'gray-200': '#D0D0D0',
};
export const lum = (hex) => {
  const c = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255).map((v) => (v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4));
  return 0.2126 * c[0] + 0.7152 * c[1] + 0.0722 * c[2];
};
export const ratio = (a, b) => { const [x, y] = [lum(HEX[a]), lum(HEX[b])].sort((m, n) => n - m); return (x + 0.05) / (y + 0.05); };
export const td = { padding: '10px 16px 10px 0', borderBottom: '1px solid var(--line)', verticalAlign: 'middle', fontSize: 14 };
export const Badge = ({ r }) => {
  const [txt, bg, fg] = r >= 4.5 ? ['AA', '#F0FFE0', '#000'] : r >= 3 ? ['Large text only', '#FDF3EC', '#B64407'] : ['Fail', '#FFF1F1', '#A2191F'];
  return <span style={{ display: 'inline-block', padding: '2px 8px', background: bg, color: fg, fontFamily: 'var(--font-mono)', fontSize: 12, textTransform: 'uppercase', letterSpacing: 'var(--ls-caps)', whiteSpace: 'nowrap' }}>{txt}</span>;
};
export const Pair = ({ fg, bg, use }) => {

    </tr>
  );
};

# Colors

The **palette** holds every raw colour, grouped by its role. Components use the **role tokens** (actions, data, surfaces, text…), which point to the palette. Never write a hex value in a component.

Blue and orange have the same steps:

| Step | When to use |
|---|---|
| **dark** | Pressed states · coloured text on the tint background |
| **base** | Buttons, links, accents, headings in colour |
| **light** | Text or icons **on dark backgrounds only** (toast, chart tooltip) |
| **tint** | Backgrounds: banners, tooltips, result bars, tinted fields |

## Actions

| Token | Points to | Use |
|---|---|---|
| `--primary` (+ `-dark` `-light` `-tint`) | blue | The main action of the screen · links · active states · Facturación and IRPF sections |
| `--attention` (+ `-dark` `-light` `-tint` `-line`) | orange | An action that **must not be skipped** (Guardar: if you leave, changes are lost) · pending things · Gastos and IVA sections |

Components take the same two names: `btn--primary` / `btn--attention`, `tab--attention`, `card--attention`, `list--attention`, `table--attention`, `field--attention`…

## Data

| Token | Points to | Use |
|---|---|---|
| `--income` · `--income-light` | blue | Ingresos: chart line, tooltip swatch |
| `--expense` · `--expense-light` | orange | Gastos, IVA, costs: chart line, tooltip swatch |

Actions and data point to the same hues today. Keeping them apart means the Guardar button could change colour without changing every expense in the charts. The charts use them through `--chart-1` (income) and `--chart-2` (expense).

## Surfaces, text & lines

Named by **where they are used**, so the name tells you which one to pick.

| Token | Points to | Use |
|---|---|---|
| `--canvas` | `--gray-50` | Page background behind the cards · empty table rows · bar tracks |
| `--surface` | `--white` | Cards, dialogs, inputs |
| `--fill` | `--gray-100` | Tabs, chips, textareas |
| `--line` | `--gray-100` | Dividers between rows |
| `--fill-hover` | `--gray-150` | Hover on `--fill` |
| `--fill-selected` | `--gray-200` | Selected neutral tab (period selector) |
| `--disabled` | `--gray-200` | Disabled buttons and controls — never readable text |
| `--line-strong` | `--gray-250` | Form-field borders |
| `--text-muted` | `--gray-600` | Grey text: dates, legends, captions, table headers |
| `--text` | `--black` | Main text, headings |

## Brand & states

| Token | Points to | Use |
|---|---|---|
| `--brand` · `--brand-text` | green · black | Identity: active year, page title band, brand chip. **Not** for "success". |
| `--positive-tint` | green tint | Result bar in your favour (a compensar) |
| `--error` · `--error-dark` · `--error-tint` | red | Error banners and messages |
| `--focus` | yellow | Focus ring for keyboard navigation, always with a 1px black halo |

## Contrast

Normal text needs **4.5 : 1** (WCAG AA). Large text (24px, or 18px bold) and icons need 3 : 1.

**One orange for everything.** `--orange` #B64407 passes AA as text on every light background of the app: white (5.5 : 1), the orange tint and the grey page (5.0 : 1), the grey fill (4.5 : 1). White text on it passes too (5.5 : 1). No exceptions: there is no "darker orange for text"; `--orange-dark` #8F3505 is only the pressed state of the orange buttons.
